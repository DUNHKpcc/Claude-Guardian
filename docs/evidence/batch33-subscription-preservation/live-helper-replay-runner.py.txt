"""New preservation replay of frozen reviewer inputs against an explicit live tree.

Does not regenerate model output or edit old provenance. Loads the unchanged
historical reviewer runner, replaces only its helper-loader in this process,
and reuses its exact fixtures, clocks, 15 calls, full comparisons and sentinels.
Public reports use repository-relative source hashes only. Python 3.10+ standard library.
"""
import argparse
import hashlib
import importlib.util
import json
from pathlib import Path
import sys

sys.dont_write_bytecode = True

def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

def module(name, path):
    spec = importlib.util.spec_from_file_location(name, path)
    value = importlib.util.module_from_spec(spec)
    sys.modules[name] = value
    spec.loader.exec_module(value)
    return value

def run_replay(source, packet):
    """Execute all original guarded calls; return only portable project evidence."""
    source, packet = Path(source).resolve(), Path(packet).resolve()
    if packet != source / 'docs/evidence/subscription-boundaries':
        raise AssertionError('Replay requires the original packet inside the selected project')
    original_runner = packet / 'reviewer-controls/replay.py'
    live_helper = source / 'skills/claude-guardian/scripts/interview_flow.py'
    live_policy = live_helper.with_name('traffic_policy.py')
    historical_helper = packet / 'retained/candidate/generator-inputs/runtime/interview_flow.py'
    fixtures = packet / 'retained/candidate/generator-inputs/helper-inputs/prior-pure-api-inputs.json'
    original_outputs = packet / 'retained/candidate/review-only/prior-observations/outputs.json'
    integration = packet / 'integration-manifest.json'
    if sha(integration) != 'fe36d3ca6620168a9288142b1c799cd95c802bd1ca812cc35cb4a4d8bac05f3a':
        raise SystemExit('Historical integration manifest changed')
    actual_controls = packet / 'reviewer-controls/actual-control-report.json'
    expected = json.loads(integration.read_text(encoding='utf-8'))['files']
    for path in [original_runner, historical_helper, fixtures, original_outputs, actual_controls]:
        relative = path.relative_to(packet).as_posix()
        if sha(path) != expected[relative]:
            raise SystemExit('Historical packet hash mismatch: ' + relative)
    receipts = {path.relative_to(source).as_posix():sha(path) for path in [original_runner, historical_helper, fixtures, original_outputs, actual_controls,
                    live_helper, live_policy, packet / 'integration-manifest.json']}
    runner = module('batch33_historical_subscription_replay_adapter', original_runner)
    def load_live():
        return module('batch33_current_subscription_replay_flow', live_helper)
    runner.load = load_live
    result = runner.replay()
    # The old replay already compares complete outputs to original observations.
    # Compare its complete call arguments/returns too, excluding version metadata.
    old = json.loads((packet / 'reviewer-controls/actual-control-report.json').read_text(encoding='utf-8'))
    if result['calls'] != old['calls'] or result['outputs'] != old['outputs']:
        raise AssertionError('Current-helper calls or outputs differ from retained controls')
    if result['call_count'] != 15 or result['blocked_io_attempts']:
        raise AssertionError('Unexpected count or I/O in current-helper replay')
    if any(sha(source / path) != value for path,value in receipts.items()):
        raise AssertionError('Source or packet changed during current-helper replay')
    document = {'kind':'current_frozen_helper_preservation_replay',
        'evidence_form':'sanitized_public_projection',
        'historical_runner_is_unchanged':True,'live_helper_loader_override_only':True,
        'model_prose_generated':False,'respondent_helper_calls':False,
        'source_sha256':receipts,'complete_calls_equal_historical_controls':True,
        'result':result}
    return document


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--source', type=Path, required=True)
    parser.add_argument('--packet', type=Path, required=True)
    parser.add_argument('--output', type=Path, required=True)
    args = parser.parse_args()
    if args.output.exists():
        raise SystemExit('Refusing to replace an existing replay report')
    document = run_replay(args.source, args.packet)
    raw = (json.dumps(document, ensure_ascii=True, sort_keys=True, indent=2) + '\n').encode('utf-8')
    with args.output.open('xb') as stream:
        stream.write(raw)
    print(json.dumps({'sha256':hashlib.sha256(raw).hexdigest(),
        'calls':15,'complete_calls_and_outputs_equal':True,'blocked_io_attempts':[]},ensure_ascii=True))

if __name__ == '__main__':
    main()
